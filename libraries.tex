% --- ATS text layer for headings -------------------------------------------
% The heading macros typeset employer / location / title / dates in a
% two-column tabular*, which PDF text extractors (Workday, Greenhouse and
% other ATS parsers) read as disconnected blocks, so fields land in the
% wrong entry or go blank. \resumeSubheadingContinue prints no employer
% at all, so that role's company/location parse as empty.
% accsupp attaches an /ActualText string to each heading: extractors read
% the linear text below, while the typeset page is unchanged -- same
% fonts, same layout, same spacing.
\usepackage{accsupp}
% \atsText{<text an ATS should read>}{<what is actually typeset>}
\newcommand{\atsText}[2]{%
  \BeginAccSupp{method=pdfstringdef,ActualText={#1}}%
  #2%
  \EndAccSupp{}%
}
% "Employer | Location", or just "Employer" when no location is given.
\newcommand{\atsEmployer}[2]{#1\if\relax\detokenize{#2}\relax\else\ | #2\fi}
% Employer of the most recent \resumeSubheading, reused by
% \resumeSubheadingContinue so a continued role still parses with one.
\newcommand{\atsEmployerName}{}
\newcommand{\atsEmployerLocation}{}

%-------------------------
% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
      {#1 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \atsText{\atsEmployer{#1}{#2}^^J#3 | #4}{%
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \small #3 & {\small #4} \\
    \end{tabular*}}\vspace{-7pt}
    \gdef\atsEmployerName{#1}%
    \gdef\atsEmployerLocation{#2}%
}

\newcommand{\resumeSubheadingContinue}[2]{
  \vspace{2pt}
    \atsText{\atsEmployer{\atsEmployerName}{\atsEmployerLocation}^^J#1 | #2}{%
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      #1 & {\small #2} \\
    \end{tabular*}}\vspace{-5pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[3]{

    \end{tabular*}}\vspace{-7pt}
